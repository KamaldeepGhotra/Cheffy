# Cheffy Core Loop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a working skeleton of the core loop — add groceries to inventory (with fuzzy-matched canonical ingredients), search for a recipe via Gemini, and generate a weekly grocery list that accounts for what's already on hand.

**Architecture:** FastAPI backend with SQLAlchemy models, backed by SQLite locally (swap to Supabase Postgres in the deployment plan — same SQLAlchemy interface, only the connection string changes). A single reusable ingredient fuzzy-matching module is shared by both the inventory-add flow and the recipe-import flow, so "chicken" always resolves to the same canonical row. A minimal React (Vite) frontend wires the three core screens to the API. No auth yet — a hardcoded `household_id` string stands in until the deployment plan adds Supabase auth.

**Tech Stack:** Python 3.11+, FastAPI, SQLAlchemy, SQLite (dev), rapidfuzz (fuzzy matching), google-generativeai (Gemini), pytest + httpx (testing), React + Vite (frontend).

**Spec:** [docs/superpowers/specs/2026-09-10-meal-prep-app-design.md](../specs/2026-09-10-meal-prep-app-design.md)

## Known Gaps (deferred, not part of this plan)

- Spec's error-handling requirement ("Gemini call fails, show error, allow retry, don't crash") is only partially covered: a failed/malformed Gemini response will currently surface as a raw 500 from the backend and an unhandled promise rejection in the frontend. Proper user-facing error/retry UI is deferred to a follow-up task.
- Spec's "ambiguous match → let user manually confirm" flow is simplified here to a fixed similarity threshold (auto-reuse if score >= 90, else auto-create) with no manual confirmation step. Revisit if auto-matching produces bad merges/duplicates in practice.

## Global Constraints

- No AI-authorship attribution anywhere in this repo (commit messages, code comments, README) — commits should read as authored by the two project owners, not an AI tool.
- Backend and frontend are separate top-level directories: `backend/` and `frontend/`.
- Grocery-list aggregation in this plan sums raw quantities per ingredient without unit conversion (e.g. it will not convert cups to grams) — this is a known, documented simplification; a full unit-conversion system is out of scope for this plan.
- Gemini API key is read from the `GEMINI_API_KEY` environment variable; never hardcoded.

```python
# backend/app/matching.py
from rapidfuzz import fuzz
from sqlalchemy.orm import Session
from app.models import Ingredient


def _normalize(name: str) -> str:
    return name.strip().lower()


def find_ingredient_matches(db: Session, name: str, limit: int = 5) -> list[tuple[Ingredient, float]]:
    normalized = _normalize(name)
    candidates = db.query(Ingredient).all()

    scored = [
        (candidate, fuzz.token_sort_ratio(normalized, _normalize(candidate.name)))
        for candidate in candidates
    ]
    scored.sort(key=lambda pair: pair[1], reverse=True)
    return scored[:limit]


def resolve_or_create_ingredient(
    db: Session,
    name: str,
    category: str | None = None,
    default_unit: str | None = None,
    threshold: float = 90.0,
) -> Ingredient:
    matches = find_ingredient_matches(db, name, limit=1)

    if matches and matches[0][1] >= threshold:
        return matches[0][0]

    new_ingredient = Ingredient(name=_normalize(name), category=category, default_unit=default_unit)
    db.add(new_ingredient)
    db.commit()
    db.refresh(new_ingredient)
    return new_ingredient
```

```python
# backend/app/routers/inventory.py
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from app.db import get_db
from app.matching import resolve_or_create_ingredient
from app.models import InventoryItem
from app.schemas import InventoryItemCreate, InventoryItemOut

router = APIRouter(prefix="/inventory", tags=["inventory"])


def _to_out(item: InventoryItem) -> InventoryItemOut:
    return InventoryItemOut(
        id=item.id,
        ingredient_id=item.ingredient_id,
        ingredient_name=item.ingredient.name,
        household_id=item.household_id,
        quantity=item.quantity,
        unit=item.unit,
        added_date=item.added_date,
    )


@router.post("", response_model=InventoryItemOut, status_code=201)
def add_inventory_item(payload: InventoryItemCreate, db: Session = Depends(get_db)):
    ingredient = resolve_or_create_ingredient(db, payload.ingredient_name)
    item = InventoryItem(
        ingredient_id=ingredient.id,
        household_id=payload.household_id,
        quantity=payload.quantity,
        unit=payload.unit,
    )
    db.add(item)
    db.commit()
    db.refresh(item)
    return _to_out(item)


@router.get("", response_model=list[InventoryItemOut])
def list_inventory(household_id: str, db: Session = Depends(get_db)):
    items = db.query(InventoryItem).filter_by(household_id=household_id).all()
    return [_to_out(item) for item in items]


@router.delete("/{item_id}", status_code=204)
def delete_inventory_item(item_id: int, db: Session = Depends(get_db)):
    item = db.query(InventoryItem).filter_by(id=item_id).first()
    if not item:
        raise HTTPException(status_code=404, detail="Inventory item not found")
    db.delete(item)
    db.commit()
```

```python
            quantity=raw_ingredient["quantity"],

    return RecipeOut(
        id=recipe.id,
        name=recipe.name,
        instructions=recipe.instructions,
        calories=recipe.calories,
```

```python
# backend/app/routers/grocery.py
from collections import defaultdict
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from app.db import get_db
from app.models import RecipeIngredient, InventoryItem
from app.schemas import GroceryListRequest, GroceryListResponse, GroceryLine

router = APIRouter(tags=["grocery"])


@router.post("/grocery-list", response_model=GroceryListResponse)
def generate_grocery_list(payload: GroceryListRequest, db: Session = Depends(get_db)):
    needed_by_key: dict[tuple[int, str], float] = defaultdict(float)
    name_by_ingredient_id: dict[int, str] = {}

    for recipe_id in payload.recipe_ids:
        servings = payload.servings.get(recipe_id, 1)
        recipe_ingredients = db.query(RecipeIngredient).filter_by(recipe_id=recipe_id).all()
        for ri in recipe_ingredients:
            key = (ri.ingredient_id, ri.unit)
            needed_by_key[key] += ri.quantity * servings
            name_by_ingredient_id[ri.ingredient_id] = ri.ingredient.name

    on_hand_by_key: dict[tuple[int, str], float] = defaultdict(float)
    inventory_items = db.query(InventoryItem).filter_by(household_id=payload.household_id).all()
    for item in inventory_items:
        on_hand_by_key[(item.ingredient_id, item.unit)] += item.quantity
        name_by_ingredient_id.setdefault(item.ingredient_id, item.ingredient.name)

    have: list[GroceryLine] = []
    need: list[GroceryLine] = []

    for key, needed_qty in needed_by_key.items():
        ingredient_id, unit = key
        on_hand_qty = on_hand_by_key.get(key, 0.0)
        name = name_by_ingredient_id[ingredient_id]

        if on_hand_qty >= needed_qty:
            have.append(GroceryLine(ingredient_name=name, needed=needed_qty, unit=unit))
        else:
            remaining = needed_qty - on_hand_qty
            need.append(GroceryLine(ingredient_name=name, needed=remaining, unit=unit))

    return GroceryListResponse(have=have, need=need)
```

```jsx
  const [quantity, setQuantity] = useState('')
  const [unit, setUnit] = useState('')

  }, [])

    } else {
      setSuggestions([])
```
